import { TkBadge } from '@takeoff-ui/react';

### Props {#divider-props}

| Name                                                                             | Type                                  | Default      | Description                                                   |
| -------------------------------------------------------------------------------- | ------------------------------------- | ------------ | ------------------------------------------------------------- |
| <TkBadge label="dataTestid" variant="primary" size="large" type="filledlight"/>  | <code>string</code>                   | null         | Sets the data-testid attribute on the root container element. |
| <TkBadge label="mx" variant="primary" size="large" type="filledlight"/>          | <code>number, string</code>           | null         | Controls vertical spacing                                     |
| <TkBadge label="my" variant="primary" size="large" type="filledlight"/>          | <code>number, string</code>           | null         | Controls horizontal spacing                                   |
| <TkBadge label="orientation" variant="primary" size="large" type="filledlight"/> | <code>"horizontal", "vertical"</code> | 'horizontal' | Controls the orientation of the divider component.            |
