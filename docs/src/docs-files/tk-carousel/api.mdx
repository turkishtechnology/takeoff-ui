import { TkBadge } from '@takeoff-ui/react';

### Props {#carousel-props}

| Name                                                                                     | Type                                                         | Default       | Description                                                                                                                                    |
| ---------------------------------------------------------------------------------------- | ------------------------------------------------------------ | ------------- | ---------------------------------------------------------------------------------------------------------------------------------------------- |
| <TkBadge label="autoplay" variant="primary" size="large" type="filledlight"/>            | <code>boolean</code>                                         | false         | Controls whether the carousel should autoplay                                                                                                  |
| <TkBadge label="autoplayDelay" variant="primary" size="large" type="filledlight"/>       | <code>number</code>                                          | 3000          | Controls the interval of the autoplay in milliseconds                                                                                          |
| <TkBadge label="circular" variant="primary" size="large" type="filledlight"/>            | <code>boolean</code>                                         | true          | Controls whether it should loop back to the start after reaching the end                                                                       |
| <TkBadge label="dataTestid" variant="primary" size="large" type="filledlight"/>          | <code>string</code>                                          | null          | Sets the data-testid attribute on the root container element.                                                                                  |
| <TkBadge label="itemsPerView" variant="primary" size="large" type="filledlight"/>        | <code>number</code>                                          | 1             | Number of items to show per view                                                                                                               |
| <TkBadge label="navigationPlacement" variant="primary" size="large" type="filledlight"/> | <code>"inside", "outside"</code>                             | 'inside'      | Placement of the navigation indicators                                                                                                         |
| <TkBadge label="navigationPosition" variant="primary" size="large" type="filledlight"/>  | <code>"bottom", "distributed", "left", "right", "top"</code> | 'distributed' | Position of the navigation indicators. For horizontal orientation: 'distributed' , 'top' , 'bottom' For vertical orientation: 'left' , 'right' |
| <TkBadge label="orientation" variant="primary" size="large" type="filledlight"/>         | <code>"horizontal", "vertical"</code>                        | 'horizontal'  | Orientation of the carousel                                                                                                                    |
| <TkBadge label="showArrows" variant="primary" size="large" type="filledlight"/>          | <code>boolean</code>                                         | true          | Controls whether the navigation arrows are shown                                                                                               |
| <TkBadge label="showIndicators" variant="primary" size="large" type="filledlight"/>      | <code>boolean</code>                                         | true          | Controls whether the carousel indicators are shown                                                                                             |
| <TkBadge label="showPlayerButton" variant="primary" size="large" type="filledlight"/>    | <code>boolean</code>                                         | false         | Controls whether the pause/play button is shown                                                                                                |
| <TkBadge label="verticalViewHeight" variant="primary" size="large" type="filledlight"/>  | <code>string</code>                                          | '300px'       | Height of the carousel when orientation is vertical                                                                                            |

### Events {#carousel-events}

| Name      | Description                  |
| --------- | ---------------------------- |
| tk-change | Emitted when item is changed |
