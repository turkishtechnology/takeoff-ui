import { TkBadge } from '@takeoff-ui/react';

### Props {#breadcrumb-item-props}

| Name                                                                            | Type                              | Default | Description                                                   |
| ------------------------------------------------------------------------------- | --------------------------------- | ------- | ------------------------------------------------------------- |
| <TkBadge label="dataTestid" variant="primary" size="large" type="filledlight"/> | <code>string</code>               | null    | Sets the data-testid attribute on the root container element. |
| <TkBadge label="href" variant="primary" size="large" type="filledlight"/>       | <code>string</code>               | null    | URL for the item                                              |
| <TkBadge label="icon" variant="primary" size="large" type="filledlight"/>       | <code>IIconOptions, string</code> | null    | Icon to display alongside the label                           |
| <TkBadge label="isCurrent" variant="primary" size="large" type="filledlight"/>  | <code>boolean</code>              | false   | Indicates if the item is the last one                         |
| <TkBadge label="isExternal" variant="primary" size="large" type="filledlight"/> | <code>boolean</code>              | false   | Whether the item is an external url                           |
| <TkBadge label="label" variant="primary" size="large" type="filledlight"/>      | <code>string</code>               | null    | Label text for the breadcrumb item                            |
